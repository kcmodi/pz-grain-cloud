        async function connectHardwareScale(targetInputId) {
            const inputField = document.getElementById(targetInputId);
            if (!("serial" in navigator)) {
                alert("Web Serial API not supported in this browser. Simulating scale payload instead.");
                let simulatedWeight = (Math.random() * (2000 - 500) + 500).toFixed(0);
                inputField.value = simulatedWeight;
                inputField.classList.add('bg-emerald-100', 'text-emerald-900', 'border-emerald-500');
                setTimeout(() => inputField.classList.remove('bg-emerald-100', 'text-emerald-900', 'border-emerald-500'), 1000);
                return;
            }
            try {
                const port = await navigator.serial.requestPort(); await port.open({ baudRate: 9600 });
                alert("COM Port connection successful. Waiting for scale string payload...");
                inputField.value = "1250"; inputField.classList.add('bg-emerald-100'); port.close();
            } catch (err) { alert("Scale connection cancelled or failed: " + err); }
        }

        function getPaymentPayload(mode, requiredTotal) {
            let method = '';
            if (mode === 'create') { let createEl = document.getElementById('create-pay-method'); method = createEl ? createEl.value : 'Cash'; }
            else if (mode === 'edit') method = document.getElementById('edit-pay-method').value;
            else if (mode === 'pending') method = document.getElementById('pending-settle-method').value;

            if (method === 'Split') {
                let sArray = ['cash','upi','card','bank','check'].map(k => {
                    let mapMethod = k === 'check' ? 'Check' : (k === 'bank' ? 'Net Banking' : (k === 'upi' ? 'UPI' : (k === 'card' ? 'Card' : 'Cash')));
                    return { m: mapMethod, a: parseFloat(document.getElementById(`split-${mode}-${k}`).value) || 0 };
                });
                let totalSplit = sArray.reduce((sum, i) => sum + i.a, 0);
                if (Math.abs(totalSplit - requiredTotal) > 0.05) { alert(`Split payment amounts must exactly equal the Grand Total of ₹${requiredTotal.toFixed(2)}.\n\nYour current split total is: ₹${totalSplit.toFixed(2)}`); return null; }
                return { method: 'Split', split_payments: sArray.filter(i => i.a > 0).map(i => ({method: i.m, amount: i.a})) };
            }
            return { method: method, split_payments: null };
        }

        function autoFillLocation(prefix) {
            clearTimeout(locationTypingTimer);
            let cityField = document.getElementById(`${prefix}-city`).value.trim();
            if (cityField.length < 2) return;
            const locationGeoMap = { "bear": { state: "DE", zip: "19701", country: "USA" }, "newark": { state: "DE", zip: "19711", country: "USA" }, "wilmington": { state: "DE", zip: "19801", country: "USA" }, "middletown": { state: "DE", zip: "19709", country: "USA" }, "ahmedabad": { state: "GJ", zip: "380001", country: "India" }, "surat": { state: "GJ", zip: "395003", country: "India" } };
            let lowerCity = cityField.toLowerCase();
            if(locationGeoMap[lowerCity]) { document.getElementById(`${prefix}-state`).value = locationGeoMap[lowerCity].state; document.getElementById(`${prefix}-zipcode`).value = locationGeoMap[lowerCity].zip; document.getElementById(`${prefix}-country`).value = locationGeoMap[lowerCity].country; return; }

            locationTypingTimer = setTimeout(async () => {
                try {
                    const response = await fetch(`https://nominatim.openstreetmap.org/search?city=${encodeURIComponent(cityField)}&country=usa&format=json&addressdetails=1&limit=1`);
                    const data = await response.json();
                    if (data && data.length > 0 && data[0].address) {
                        const address = data[0].address;

                        document.getElementById(`${prefix}-country`).value = "USA";
                    }
                } catch (error) {}
            }, 1000);
        }

        function downloadPDFReceipt() {

        async function submitNewContract() {
            let payload = {
                customer_code: document.getElementById('contract-customer').value,
                product_code: document.getElementById('contract-product').value,
                qty: document.getElementById('contract-qty').value,
                rate: document.getElementById('contract-rate').value
            };

            if(!payload.customer_code || !payload.product_code || !payload.qty || !payload.rate) {
                return alert("Please fill out all contract parameters.");
            }

            try {
                let r = await fetch('/api/create-contract', { 
                    method: 'POST', 
                    headers: {'Content-Type': 'application/json'}, 
                    body: JSON.stringify(payload) 
                });
                let d = await r.json();
                
                if(d.success) {
                    alert(d.message);
                    closeContractModal();
                    loadContracts(); // Refresh table live
                } else {
                    alert("❌ Error: " + d.message);
                }
            } catch(e) {
                alert("Error connecting to server.");
            }
        }

        async function loadContracts() {
            let tbody = document.getElementById('contracts-table-body');
            if(!tbody) return;
            tbody.innerHTML = '<tr><td colspan="6" class="p-4 text-center text-xs font-bold text-slate-500">Loading contracts...</td></tr>';
            
            try {
                let r = await fetch('/api/get-contracts');
                let d = await r.json();
                
                if(d.success) {
                    tbody.innerHTML = '';
                    if(d.contracts.length === 0) {
                        tbody.innerHTML = '<tr><td colspan="6" class="p-8 text-center text-xs font-bold text-slate-500">No active contracts found in the ledger.</td></tr>';
                        return;
                    }
                    
                    d.contracts.forEach(c => {
                        let statusColor = c.status === 'Active' ? 'text-emerald-700 dark:text-emerald-400 bg-emerald-100 dark:bg-emerald-500/10 border-emerald-300 dark:border-emerald-500/20' : 'text-slate-700 dark:text-slate-400 bg-slate-200 dark:bg-white/5 border-slate-300 dark:border-white/10';
                        
                        tbody.innerHTML += `
                        <tr class="hover:bg-slate-50 dark:hover:bg-white/[0.02] transition-colors">
                            <td class="p-3 font-mono font-black text-indigo-700 dark:text-amber-500">${c.id}</td>
                            <td class="p-3 font-extrabold text-slate-900 dark:text-slate-200">${c.customer}</td>
                            <td class="p-3 font-bold text-slate-600 dark:text-slate-400">${c.product}</td>
                            <td class="p-3 font-black text-slate-900 dark:text-white">${c.qty.toLocaleString()} kg</td>
                            <td class="p-3 font-mono font-black text-emerald-600 dark:text-emerald-400">₹${c.rate.toFixed(2)}</td>
                            <td class="p-3 text-right">
                                <span class="px-2 py-1 rounded text-[9px] uppercase font-black border ${statusColor}">${c.status}</span>
                            </td>
                        </tr>`;
                    });
                }
            } catch(e) {
                tbody.innerHTML = '<tr><td colspan="6" class="p-4 text-center text-xs font-bold text-rose-500">Failed to load contracts.</td></tr>';
            }
        }
